12/// Represents the address of an index in the database.
13///
14/// An address has a string *name* and an optional byte *key*. An index is uniquely identified
15/// by its address. Different addresses correspond to different indexes. Addresses with the same
16/// name and differing keys are said to belong to the same *group* (see also [`Group`]). Groups
17/// can be used for a potentially unbounded group of indexes that can be identified by a certain
18/// key (for example, `ListIndex` with the transaction history of a wallet keyed by the
19/// `PublicKey` of the wallet).
20///
21/// In contrast with [`ResolvedAddress`], `IndexAddress` is a high-level logical construct;
22/// it does not directly map to key-value storage abstractions (column families and their keys).
23///
24/// [`Group`]: indexes/group/struct.Group.html
25/// [`ResolvedAddress`]: struct.ResolvedAddress.html
26///
27/// # Examples
28///
29/// `IndexAddress` can be used implicitly, since `&str` and `(&str, &impl BinaryKey)` can both
30/// be converted into an address.
31///
32/// ```
33/// use matterdb::{access::CopyAccessExt, IndexAddress, TemporaryDB, Database};
34///
35/// let db = TemporaryDB::new();
36/// let fork = db.fork();
37///
38/// // Using a string address:
39/// let map = fork.get_map::<_, String, u8>("map");
40/// // Using an address within an index family:
41/// let list = fork.get_list::<_, String>(("index", &3_u32));
42/// // Using `IndexAddress` explicitly:
43/// let addr = IndexAddress::from_root("data").append_key(&vec![1, 2, 3]);
44/// let set = fork.get_key_set::<_, u64>(addr);
45/// ```
46#[derive(Debug, Clone, Eq, PartialEq, Hash, Default)]
47pub struct IndexAddress {
48    pub(super) name: String,
49    pub(super) id_in_group: Option<Vec<u8>>,
50    pub(super) in_migration: bool,
51}

53impl IndexAddress {

73    /// Prepends a name part to `IndexAddress`. The name is separated from the existing name
74    /// by a dot `.`.
75    ///
76    /// # Examples
77    ///
78    /// ```
79    /// # use matterdb::IndexAddress;
80    /// let addr = IndexAddress::from_root("foo");
81    /// let prefixed = addr.prepend_name("prefix");
82    /// assert_eq!(prefixed.name(), "prefix.foo");
83    /// ```
84    #[must_use]
85    pub fn prepend_name(self, prefix: &str) -> Self {
86        let name = if self.name.is_empty() {
87            prefix.to_owned()
88        } else {
89            // Because `concat` is faster than `format!("...")` in all cases.
90            [prefix, ".", self.name()].concat()
91        };
92
93        Self { name, ..self }
94    }
95
96    /// Appends a name part to `IndexAddress`. The name is separated from the existing name
97    /// by a dot `.`.
98    ///
99    /// # Examples
100    ///
101    /// ```
102    /// # use matterdb::IndexAddress;
103    /// let addr = IndexAddress::from_root("foo");
104    /// let suffixed = addr.append_name("suffix");
105    /// assert_eq!(suffixed.name(), "foo.suffix");
106    /// ```
107    #[must_use]
108    pub fn append_name(self, suffix: &str) -> Self {
109        let name = if self.name.is_empty() {
110            suffix.to_owned()
111        } else {
112            // Because `concat` is faster than `format!("...")` in all cases.
113            [self.name(), ".", suffix].concat()
114        };
115
116        Self { name, ..self }
117    }
118
119    /// Appends a key to the `IndexAddress`.
120    #[must_use]
121    pub fn append_key<K: BinaryKey + ?Sized>(self, suffix: &K) -> Self {
122        let bytes = if let Some(ref bytes) = self.id_in_group {
123            concat_keys!(bytes, suffix)
124        } else {
125            concat_keys!(suffix)
126        };
127
128        Self {
129            id_in_group: Some(bytes),
130            ..self
131        }
132    }

134    pub(crate) fn set_in_migration(&mut self) {
135        self.in_migration = true;
136    }

138    /// Full address with a separator between `name` and `bytes` represented as byte array.
139    pub(crate) fn fully_qualified_name(&self) -> Vec<u8> {
140        /// Separator between the name and the additional bytes in family indexes.
141        const INDEX_NAME_SEPARATOR: &[u8] = &[SEPARATOR_CHAR];
142        const MIGRATION_PREFIX: &[u8] = &[MIGRATION_CHAR];
143
144        match (self.in_migration, self.id_in_group()) {
145            (true, Some(bytes)) => {
146                concat_keys!(MIGRATION_PREFIX, self.name(), INDEX_NAME_SEPARATOR, bytes)
147            }
148            (false, Some(bytes)) => concat_keys!(self.name(), INDEX_NAME_SEPARATOR, bytes),
149            (true, None) => concat_keys!(MIGRATION_PREFIX, self.name()),
150            (false, None) => self.name.as_bytes().to_vec(),
151        }
152    }
153
154    /// Returns the common prefix of fully qualified names for the child indexes.
155    pub(crate) fn qualified_prefix(&self) -> Vec<u8> {
156        let mut prefix = self.fully_qualified_name();
157        if self.id_in_group.is_none() {
158            prefix.push(SEPARATOR_CHAR);
159        }
160        prefix
161    }
162
163    /// Infers the name part of the fully qualified name that was obtained with
164    /// `fully_qualified_name`. This is the part corresponding to `ResolvedAddress.name`.
165    /// `min_name_len` specifies the minimum known length of the name part.
166    ///
167    /// # Return value
168    ///
169    /// Returns the restored name part of the index address and a bool flag indicating whether
170    /// the index is a part of a group.
171    #[allow(unsafe_code)]
172    pub(super) fn parse_fully_qualified_name(
173        qualified_name: &[u8],
174        min_name_len: usize,
175    ) -> (String, bool) {
176        let (cutoff_index, is_in_group) = qualified_name[min_name_len..]
177            .iter()
178            .position(|&byte| byte == SEPARATOR_CHAR)
179            .map_or_else(
180                || (qualified_name.len(), false),
181                |pos| (pos + min_name_len, true),
182            );
183
184        let name = if qualified_name[0] == MIGRATION_CHAR {
185            // The `^` initial char is not mapped to the resolved CF name.
186            qualified_name[1..cutoff_index].to_vec()
187        } else {
188            qualified_name[..cutoff_index].to_vec()
189        };
190        debug_assert!(
191            name.is_ascii(),
192            "BUG: non-ASCII chars in the name part of a fully qualified index name \
193             (qualified name = {:?}, inferred name part = {})",
194            qualified_name,
195            String::from_utf8_lossy(&name)
196        );
197
198        let name = unsafe {
199            // SAFETY:
200            // Safe by construction; metadata keys before the `\0` separator
201            // correspond to the index names, which consist of a subset of
202            // ASCII chars.
203            String::from_utf8_unchecked(name)
204        };
205        (name, is_in_group)
206    }
207
208    /// Converts a migration namespace into the form that all indexes in the namespace
209    /// begin with.
210    #[inline]
211    pub(super) fn qualify_migration_namespace(namespace: &str) -> Vec<u8> {
212        ["^", namespace, "."].concat().into_bytes()
213    }
214
215    #[inline]
216    pub(super) fn migrate_qualified_name(qualified_name: &[u8]) -> &[u8] {
217        debug_assert_eq!(
218            qualified_name[0], MIGRATION_CHAR,
219            "Qualified name {qualified_name:?} is not in migration"
220        );
221        &qualified_name[1..]
222    }

247/// Resolved address of a view.
248///
249/// While an [`IndexAddress`] is a logical location of a view, a `ResolvedAddress`
250/// represents its location in the key-value storage. The mapping between `IndexAddress`es
251/// and `ResolvedAddress`es is internal to the database logic.
252///
253/// [`IndexAddress`]: struct.IndexAddress.html
254#[derive(Debug, Clone, Hash, PartialEq, Eq)]
255pub struct ResolvedAddress {
256    /// Name of the column family where the view is stored.
257    pub name: String,
258    /// Index identifier within a column family. If set to `None`, then the column family contains
259    /// a single view. If `id` is `Some(_)`, then its value should be used to split key spaces
260    /// for different views in the same column family. In other words, key spaces for two addresses
261    /// with equal `name` and `id`s `Some(x)` and `Some(y)`, `x != y`, must not intersect.
262    pub id: Option<NonZeroU64>,
263}
264
265impl ResolvedAddress {
266    pub(crate) fn new(name: impl Into<String>, id: Option<NonZeroU64>) -> Self {
267        Self {
268            name: name.into(),
269            id,
270        }
271    }
272
273    /// Creates a system view. System views are low-level (i.e., they are not wrapped in indexes).
274    pub(crate) fn system(name: impl Into<String>) -> Self {
275        Self::new(name, None)
276    }
277
278    pub(crate) fn id_to_bytes(&self) -> Option<[u8; 8]> {
279        self.id.map(|id| id.get().to_le_bytes())
280    }
281
282    /// Returns `key` prefixed by the `id`.
283    pub(crate) fn keyed<'k>(&self, key: &'k [u8]) -> Cow<'k, [u8]> {
284        match self.id {
285            None => Cow::Borrowed(key),
286            Some(id) => {
287                let mut bytes = Vec::with_capacity(8 + key.len());
288                bytes.extend_from_slice(&id.get().to_le_bytes());
289                bytes.extend_from_slice(key);
290                bytes.into()
291            }
292        }
293    }
294}

304#[test]
305fn address_resolution() {
306    {
307        let (name, is_in_group) = IndexAddress::parse_fully_qualified_name(b"some.list", 0);
308        assert_eq!(name, "some.list");
309        assert!(!is_in_group);
310    }
311    {
312        let (name, is_in_group) = IndexAddress::parse_fully_qualified_name(b"some.list", 9);
313        assert_eq!(name, "some.list");
314        assert!(!is_in_group);
315    }
316    {
317        let (name, is_in_group) =
318            IndexAddress::parse_fully_qualified_name(b"some.list\0key\xa0", 9);
319        assert_eq!(name, "some.list");
320        assert!(is_in_group);
321    }
