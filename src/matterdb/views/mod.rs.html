82/// Allows to read data from the database. The data consists of a snapshot and
83/// changes relative to this snapshot. Depending on the implementation, the changes
84/// can be empty, immutable or mutable.
85///
86/// This trait is rarely needs to be used directly; [`Access`] is a more high-level trait
87/// encompassing access to database. In particular, using `snapshot()` method to convert
88/// the implementation into `&dyn Snapshot` is logically incorrect, because the snapshot
89/// may not reflect the most recent state of `RawAccess`.
90///
91/// [`Access`]: trait.Access.html
92pub trait RawAccess: Clone {
93    /// Type of the `changes()` that will be applied to the database.
94    type Changes: ChangeSet;
95
96    /// Reference to a `Snapshot`. This is the base relative to which the changes are defined.
97    fn snapshot(&self) -> &dyn Snapshot;
98    /// Returns changes related to specific `address` compared to the `snapshot()`.
99    fn changes(&self, address: &ResolvedAddress) -> Self::Changes;
100}

102/// Allows to mutate data in indexes.
103///
104/// This is a marker trait that is used as a bound for mutable operations on indexes.
105/// It can be used in the same way for high-level database objects:
106///
107/// # Example
108///
109/// ```
110/// use matterdb::{access::{Access, RawAccessMut}, ListIndex, MapIndex};
111///
112/// pub struct Schema<T: Access> {
113///     list: ListIndex<T::Base, String>,
114///     map: MapIndex<T::Base, u64, u64>,
115/// }
116///
117/// impl<T: Access> Schema<T>
118/// where
119///     T::Base: RawAccessMut,
120/// {
121///     pub fn mutate(&mut self) {
122///         self.list.push("foo".to_owned());
123///         self.map.put(&1, 2);
124///     }
125/// }
126/// ```
127pub trait RawAccessMut: RawAccess {}
128
129impl<'a, T> RawAccessMut for T where T: RawAccess<Changes = ChangesMut<'a>> {}

131/// Converts index access to a readonly presentation. The conversion operation is cheap.
132pub trait AsReadonly: RawAccess {

136    /// Performs the conversion.

140macro_rules! impl_snapshot_access {
141    ($typ:ty) => {
142        impl RawAccess for $typ {
143            type Changes = ();
144
145            fn snapshot(&self) -> &dyn Snapshot {
146                self.as_ref()
147            }
148
149            fn changes(&self, _address: &ResolvedAddress) -> Self::Changes {}
150        }
151
152        impl AsReadonly for $typ {
153            type Readonly = Self;
154
155            fn as_readonly(&self) -> Self::Readonly {
156                self.clone()
157            }
158        }
159    };
160}

167impl<T: RawAccess> ViewInner<T> {
168    fn snapshot(&self) -> &dyn Snapshot {
169        self.index_access.snapshot()
170    }
171
172    fn get_bytes(&self, key: &[u8]) -> Option<Vec<u8>> {
173        self.changes
174            .as_ref()
175            .map_or(Err(()), |changes| changes.get(key))
176            // At this point, `Err(_)` signifies that we need to retrieve data from the snapshot.
177            .unwrap_or_else(|()| self.snapshot().get(&self.address, key))
178    }
179
180    fn contains_raw_key(&self, key: &[u8]) -> bool {
181        self.changes
182            .as_ref()
183            .map_or(Err(()), |changes| changes.contains(key))
184            // At this point, `Err(_)` signifies that we need to retrieve data from the snapshot.
185            .unwrap_or_else(|()| self.snapshot().contains(&self.address, key))
186    }
187
188    fn iter_bytes(&self, from: &[u8]) -> BytesIter<'_> {
189        use std::ops::Bound::{Included, Unbounded};
190
191        let changes_iter = self
192            .changes
193            .as_ref()
194            .map(|changes| changes.data.range::<[u8], _>((Included(from), Unbounded)));
195
196        let is_cleared = self.changes.as_ref().is_some_and(ViewChanges::is_cleared);
197        if is_cleared {
198            // Ignore all changes from the snapshot.
199            Box::new(ChangesIter::new(changes_iter.unwrap()))
200        } else {
201            Box::new(ForkIter::new(
202                self.snapshot().iter(&self.address, from),
203                changes_iter,
204            ))
205        }
206    }
207}

209impl<T: RawAccess> View<T> {
210    /// Creates a new view for an index with the specified address.
211    pub(crate) fn new(index_access: T, address: impl Into<ResolvedAddress>) -> Self {
212        let address = address.into();
213        let changes = index_access.changes(&address);
214        Self::Real(ViewInner {
215            address,
216            index_access,
217            changes,
218        })
219    }
220
221    /// Creates a new phantom view. The phantom views do not borrow changes and do not retain
222    /// resolved address / access.
223    pub(crate) fn new_phantom() -> Self {
224        Self::Phantom
225    }
226
227    /// Returns the access this view is attached to. If this view is phantom, returns `None`.
228    pub(crate) fn access(&self) -> Option<&T> {
229        match self {
230            Self::Real(ViewInner { index_access, .. }) => Some(index_access),
231            Self::Phantom => None,
232        }
233    }
234
235    fn get_bytes(&self, key: &[u8]) -> Option<Vec<u8>> {
236        match self {
237            Self::Real(inner) => inner.get_bytes(key),
238            Self::Phantom => None,
239        }
240    }
241
242    fn contains_raw_key(&self, key: &[u8]) -> bool {
243        match self {
244            Self::Real(inner) => inner.contains_raw_key(key),
245            Self::Phantom => false,
246        }
247    }
248
249    fn iter_bytes(&self, from: &[u8]) -> BytesIter<'_> {
250        match self {
251            Self::Real(inner) => inner.iter_bytes(from),
252            Self::Phantom => Box::new(EmptyIterator),
253        }
254    }
255
256    /// Returns a value of *any* type corresponding to the key of *any* type.
257    pub(crate) fn get<K, V>(&self, key: &K) -> Option<V>
258    where
259        K: BinaryKey + ?Sized,
260        V: BinaryValue,
261    {
262        self.get_bytes(&key_bytes(key)).map(|v| {
263            BinaryValue::from_bytes(Cow::Owned(v)).expect("Error while deserializing value")
264        })
265    }
266
267    /// Returns `true` if the index contains a value of *any* type for the specified key of
268    /// *any* type.
269    pub(crate) fn contains<K>(&self, key: &K) -> bool
270    where
271        K: BinaryKey + ?Sized,
272    {
273        self.contains_raw_key(&key_bytes(key))
274    }
275
276    /// Returns an iterator over the entries of the index in ascending order. The iterator element
277    /// type is *any* key-value pair. An argument `subprefix` allows specifying a subset of keys
278    /// for iteration.
279    pub(crate) fn iter<P, K, V>(&self, subprefix: &P) -> Iter<'_, K, V>
280    where
281        P: BinaryKey + ?Sized,
282        K: BinaryKey + ?Sized,
283        V: BinaryValue,
284    {
285        let iter_prefix = key_bytes(subprefix);
286        Iter {
287            base_iter: self.iter_bytes(&iter_prefix),
288            prefix: iter_prefix,
289            detach_prefix: false,
290            ended: false,
291            _k: PhantomData,
292            _v: PhantomData,
293        }
294    }
295
296    /// Returns an iterator over the entries of the index in ascending order starting from the
297    /// specified key. The iterator element type is *any* key-value pair. An argument `subprefix`
298    /// allows specifying a subset of iteration.
299    pub(crate) fn iter_from<P, F, K, V>(&self, subprefix: &P, from: &F) -> Iter<'_, K, V>
300    where
301        P: BinaryKey,
302        F: BinaryKey + ?Sized,
303        K: BinaryKey + ?Sized,
304        V: BinaryValue,
305    {
306        let iter_prefix = key_bytes(subprefix);
307        let iter_from = key_bytes(from);
308        Iter {
309            base_iter: self.iter_bytes(&iter_from),
310            prefix: iter_prefix,
311            detach_prefix: false,
312            ended: false,
313            _k: PhantomData,
314            _v: PhantomData,
315        }
316    }
317
318    /// Sets a key / value pair in the view storage, unless the view is backed by a readonly access
319    /// (in which case, the changes are forgotten).
320    ///
321    /// # Return value
322    ///
323    /// Returns whether the changes were saved.
324    pub(crate) fn put_or_forget<K, V>(&mut self, key: &K, value: V) -> bool
325    where
326        K: BinaryKey + ?Sized,
327        V: BinaryValue,
328    {
329        if let Self::Real(inner) = self {
330            if let Some(changes) = inner.changes.as_mut() {
331                changes
332                    .data
333                    .insert(concat_keys!(key), Change::Put(value.into_bytes()));
334                return true;
335            }
336        }
337        false
338    }
339}

341impl<T: RawAccessMut> View<T> {
342    fn changes_mut(&mut self) -> &mut ViewChanges {
343        const ACCESS_ERROR: &str = "Attempt to modify a readonly view of the database using a generic access. \
344             The caller should check the access type before calling any mutable methods";
345
346        match self {
347            Self::Real(ViewInner { changes, .. }) => changes.as_mut().expect(ACCESS_ERROR),
348            Self::Phantom => panic!("{}", ACCESS_ERROR),
349        }
350    }
351
352    /// Inserts a key-value pair into the fork.
353    pub(crate) fn put<K, V>(&mut self, key: &K, value: V)
354    where
355        K: BinaryKey + ?Sized,
356        V: BinaryValue,
357    {
358        self.changes_mut()
359            .data
360            .insert(concat_keys!(key), Change::Put(value.into_bytes()));
361    }
362
363    /// Removes a key from the view.
364    pub(crate) fn remove<K>(&mut self, key: &K)
365    where
366        K: BinaryKey + ?Sized,
367    {
368        self.changes_mut()
369            .data
370            .insert(concat_keys!(key), Change::Delete);
371    }
372
373    /// Clears the view removing all its elements.
374    pub(crate) fn clear(&mut self) {
375        self.changes_mut().clear();
376    }
377}

406        }

410impl<'a, T> BytesIterator for ChangesIter<'a, T>
411where
412    T: Iterator<Item = (&'a Vec<u8>, &'a Change)>,
413{
414    fn next(&mut self) -> Option<(&[u8], &[u8])> {
415        loop {
416            match self.inner.next() {
417                Some((key, Change::Put(value))) => {
418                    return Some((key.as_slice(), value.as_slice()));
419                }
420                Some((_, &Change::Delete)) => {}
421                None => {
422                    return None;
423                }
424            }
425        }
426    }
427
428    fn peek(&mut self) -> Option<(&[u8], &[u8])> {
429        loop {
430            match self.inner.peek() {
431                Some((key, Change::Put(value))) => {
432                    return Some((key.as_slice(), value.as_slice()));
433                }
434                Some((_, Change::Delete)) => {
435                    // Advance the iterator. Since we've already peeked the value,
436                    // we can safely drop it.
437                    self.inner.next();
438                }
439                None => {
440                    return None;
441                }
442            }
443        }
444    }
445}

501impl<K, V> Iterator for Iter<'_, K, V>
502where
503    K: BinaryKey + ?Sized,
504    V: BinaryValue,
505{
506    type Item = (K::Owned, V);
507
508    fn next(&mut self) -> Option<Self::Item> {
509        if self.ended {
510            return None;
511        }
512
513        if let Some((key_slice, value_slice)) = self.base_iter.next() {
514            if key_slice.starts_with(&self.prefix) {
515                let key = if self.detach_prefix {
516                    // Since we've checked `start_with`, slicing here cannot panic.
517                    K::read(&key_slice[self.prefix.len()..])
518                } else {
519                    K::read(key_slice)
520                };
521                let value = V::from_bytes(Cow::Borrowed(value_slice))
522                    .expect("Unable to decode value from bytes");
523                return Some((key, value));
524            }
525        }
526
527        self.ended = true;
528        None
529    }
