1//! High-level access to database.
2//!
3//! # Overview
4//!
5//! The core type in this module is the [`Access`] trait, which provides ability to access
6//! [indexes] from the database. The `Access` trait has several implementations:
7//!
8//! - `Access` is implemented for [`RawAccess`]es, that is, types that provide access to the
9//!   entire database. [`Snapshot`], [`Fork`] and [`ReadonlyFork`] fall into this category.
10//! - [`Prefixed`] restricts an access to a single *namespace*.
11//! - [`Migration`]s are used for data created during [migrations]. Similar to `Prefixed`, migrations
12//!   are separated by namespaces.
13//! - [`Scratchpad`]s can be used for temporary data. They are distinguished by namespaces as well.
14//!
15//! [`CopyAccessExt`] extends [`Access`] and provides helper methods to instantiate indexes. This
16//! is useful in quick-and-dirty testing. For more complex applications, consider deriving
17//! data schema via [`FromAccess`].
18//!
19//! # Guarantees
20//!
21//! - Namespaced accesses (`Prefixed`, `Migration`s and `Scratchpad`s) do not intersect (i.e.,
22//!   do not have common indexes) for different namespaces. They also do not intersect for
23//!   different access types, even for the same namespace; for example, a `Prefixed` access
24//!   can never access an index from a `Migration` or a `Scratchpad` and vice versa.
25//! - For all listed `Access` implementations, different addresses *within* an `Access` correspond
26//!   to different indexes.
27//! - However, if we consider multiple accesses, indexes can alias. For example, an index
28//!   with address `bar` from a `Prefixed<&Fork>` in namespace `foo` can also be accessed via
29//!   address `foo.bar` from the underlying `Fork`.
30//!
31//! [`Access`]: trait.Access.html
32//! [indexes]: ../index.html#indexes
33//! [`RawAccess`]: trait.RawAccess.html
34//! [`Snapshot`]: ../trait.Snapshot.html
35//! [`Fork`]: ../struct.Fork.html
36//! [`ReadonlyFork`]: ../struct.ReadonlyFork.html
37//! [`Prefixed`]: struct.Prefixed.html
38//! [`Migration`]: ../migration/struct.Migration.html
39//! [migrations]: ../migration/index.html
40//! [`Scratchpad`]: ../migration/struct.Scratchpad.html
41//! [`CopyAccessExt`]: trait.CopyAccessExt.html
42//! [`FromAccess`]: trait.FromAccess.html

58/// High-level access to database data.
59///
60/// This trait is not intended to be implemented by the types outside the crate; indeed,
61/// it instantiates several crate-private types. Correspondingly, `Access` methods
62/// rarely need to be used directly; use [its extension trait][`CopyAccessExt`] instead.
63///
64/// [`CopyAccessExt`]: trait.CopyAccessExt.html
65///
66/// # Examples
67///
68/// `Access` can be used as a bound on structured database objects and their
69/// readonly methods:
70///
71/// ```
72/// use matterdb::{access::Access, ListIndex, MapIndex};
73///
74/// struct Schema<T: Access> {
75///     list: ListIndex<T::Base, u64>,
76///     map: MapIndex<T::Base, str, u64>,
77/// }
78///
79/// impl<T: Access> Schema<T> {
80///     fn get_some_data(&self) -> Option<u64> {
81///         Some(self.list.get(0)? + self.map.get("foo")?)
82///     }
83/// }
84/// ```
85pub trait Access: Clone {
86    /// Raw access serving as the basis for created indexes.
87    type Base: RawAccess;
88
89    /// Gets index metadata at the specified address, or `None` if there is no index.
90    ///
91    /// # Errors
92    ///
93    /// Returns access errors.
94    fn get_index_metadata(self, addr: IndexAddress) -> Result<Option<IndexMetadata>, AccessError>;
95
96    /// Gets or creates a generic view with the specified address.
97    ///
98    /// # Errors
99    ///
100    /// Returns access errors.
101    fn get_or_create_view(
102        self,
103        addr: IndexAddress,
104        index_type: IndexType,
105    ) -> Result<ViewWithMetadata<Self::Base>, AccessError>;
106
107    /// Returns an iterator over keys in a group with the specified address.
108    ///
109    /// The iterator buffers keys in memory and may become inconsistent for accesses
110    /// based on [`ReadonlyFork`].
111    ///
112    /// [`ReadonlyFork`]: ../struct.ReadonlyFork.html
113    fn group_keys<K>(self, base_addr: IndexAddress) -> GroupKeys<Self::Base, K>
114    where
115        K: BinaryKey + ?Sized,
116        Self::Base: AsReadonly<Readonly = Self::Base>;
117}
118
119impl<T: RawAccess> Access for T {
120    type Base = Self;
121
122    fn get_index_metadata(self, addr: IndexAddress) -> Result<Option<IndexMetadata>, AccessError> {
123        ViewWithMetadata::get_metadata(self, &addr)
124    }
125
126    fn get_or_create_view(
127        self,
128        addr: IndexAddress,
129        index_type: IndexType,
130    ) -> Result<ViewWithMetadata<Self::Base>, AccessError> {
131        ViewWithMetadata::get_or_create(self, &addr, index_type)
132    }
133
134    fn group_keys<K>(self, base_addr: IndexAddress) -> GroupKeys<Self::Base, K>
135    where
136        K: BinaryKey + ?Sized,
137        Self::Base: AsReadonly<Readonly = Self::Base>,
138    {
139        GroupKeys::new(self, &base_addr)
140    }
141}

143/// Access that prepends the specified prefix to each created view. The prefix is separated
144/// from user-provided names with a dot char `'.'`.
145///
146/// Since the prefix itself cannot contain a dot, `Prefixed` accesses provide namespace
147/// separation. A set of indexes to which `Prefixed` provides access does not intersect
148/// with a set of indexes accessed by a `Prefixed` instance with another prefix. Additionally,
149/// index in `Prefixed` accesses do not intersect with indexes in special-purpose `Access`
150/// implementations ([`Migration`]s and [`Scratchpad`]s).
151///
152/// [`Migration`]: ../migration/struct.Migration.html
153/// [`Scratchpad`]: ../migration/struct.Scratchpad.html
154///
155/// # Examples
156///
157/// ```
158/// use matterdb::{access::{AccessExt, CopyAccessExt, Prefixed}, Database, TemporaryDB};
159///
160/// let db = TemporaryDB::new();
161/// let fork = db.fork();
162/// let prefixed = Prefixed::new("prefixed", &fork);
163/// prefixed.get_list("list").extend(vec![1_u32, 2, 3]);
164/// let same_list = fork.get_list::<_, u32>("prefixed.list");
165/// assert_eq!(same_list.len(), 3);
166/// ```
167#[derive(Debug, Clone)]
168pub struct Prefixed<T> {
169    access: T,
170    prefix: String,
171}
172
173// **NB.** Must not be made public! This would allow the caller to violate access restrictions
174// imposed by `Prefixed`.
175impl<T> Prefixed<T> {
176    pub(crate) fn access(&self) -> &T {
177        &self.access
178    }
179
180    pub(crate) fn into_parts(self) -> (String, T) {
181        (self.prefix, self.access)
182    }
183}
184
185impl<T: RawAccess> Prefixed<T> {
186    /// Creates a new prefixed access.
187    ///
188    /// # Panics
189    ///
190    /// - Will panic if the prefix is not a [valid prefix name].
191    ///
192    /// [valid prefix name]: ../validation/fn.is_valid_index_name_component.html
193    pub fn new(prefix: impl Into<String>, access: T) -> Self {
194        let prefix = prefix.into();
195        assert_valid_name_component(prefix.as_ref());
196        Self { access, prefix }
197    }
198}
199
200impl<T: RawAccess> Access for Prefixed<T> {
201    type Base = T;
202
203    fn get_index_metadata(self, addr: IndexAddress) -> Result<Option<IndexMetadata>, AccessError> {
204        let prefixed_addr = addr.prepend_name(self.prefix.as_ref());
205        self.access.get_index_metadata(prefixed_addr)
206    }
207
208    fn get_or_create_view(
209        self,
210        addr: IndexAddress,
211        index_type: IndexType,
212    ) -> Result<ViewWithMetadata<Self::Base>, AccessError> {
213        let prefixed_addr = addr.prepend_name(self.prefix.as_ref());
214        self.access.get_or_create_view(prefixed_addr, index_type)
215    }
216
217    fn group_keys<K>(self, base_addr: IndexAddress) -> GroupKeys<Self::Base, K>
218    where
219        K: BinaryKey + ?Sized,
220        Self::Base: AsReadonly<Readonly = Self::Base>,
221    {
222        let prefixed_addr = base_addr.prepend_name(self.prefix.as_ref());
223        self.access.group_keys(prefixed_addr)
224    }
225}

227/// Access error together with the location information.
228#[derive(Debug, Error)]
229pub struct AccessError {
230    /// Address of the index where the error has occurred.
231    pub addr: IndexAddress,
232    /// Error kind.
233    #[source]
234    pub kind: AccessErrorKind,
235}
236
237impl fmt::Display for AccessError {
238    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
239        // TODO: implement `Display` for `IndexAddress` for human-readable errors
240        write!(formatter, "Error accessing {:?}: {}", self.addr, self.kind)
241    }
242}

244/// Error that can be emitted during accessing an object from the database.
245#[derive(Debug, Error)]
246#[non_exhaustive]
247pub enum AccessErrorKind {
248    /// Index has wrong type.
249    #[error("Wrong index type: expected {:?}, but got {:?}", expected, actual)]
250    WrongIndexType {
251        /// Expected index type.
252        expected: IndexType,
253        /// Actual index type.
254        actual: IndexType,
255    },
256
257    /// Index name is reserved. It's forbidden for user to create indexes with names
258    /// starting with `__` and not containing a dot `.`.
259    #[error("Index name is reserved")]
260    ReservedName,
261
262    /// Index name is empty.
263    #[error("Index name must not be empty")]
264    EmptyName,
265
266    /// Index contains invalid characters.
267    #[error("Invalid characters used in name ({}). Use {}", name, allowed_chars)]
268    InvalidCharsInName {
269        /// Name that contains invalid chars.
270        name: String,
271        /// Characters allowed in name.
272        allowed_chars: &'static str,
273    },
274
275    /// Invalid tombstone location.
276    #[error("Invalid tombstone location. Tombstones can only be created in migrations")]
277    InvalidTombstone,
278
279    /// Custom error.
280    #[error("{0}")]
281    Custom(#[source] anyhow::Error),
282}

284/// Constructs an object atop the database. The constructed object provides access to data
285/// in the DB, akin to an object-relational mapping.
286///
287/// The access to DB can be readonly or read-write, depending on the `T: Access` type param.
288/// Most object should implement `FromAccess<T>` for all `T: Access`.
289///
290/// Simplest `FromAccess` implementors are indexes; it is also implemented for [`Lazy`] and [`Group`].
291/// `FromAccess` can be implemented for more complex *components*. Thus, `FromAccess` can
292/// be used to compose storage objects from simpler ones.
293///
294/// [`Lazy`]: ../struct.Lazy.html
295/// [`Group`]: ../indexes/group/struct.Group.html
296///
297/// # Examples
298///
299/// Component with two inner indexes. `FromAccess` is automatically derived using
300/// the `matterdb_derive` crate.
301///
302/// ```
303/// use matterdb_derive::FromAccess;
304/// # use matterdb::{
305/// #     access::{Access, CopyAccessExt, AccessError, FromAccess, RawAccessMut},
306/// #     Database, Entry, Group, Lazy, MapIndex, IndexAddress, TemporaryDB,
307/// # };
308///
309/// #[derive(FromAccess)]
310/// struct InsertOnlyMap<T: Access> {
311///     map: MapIndex<T::Base, str, String>,
312///     len: Entry<T::Base, u64>,
313/// }
314///
315/// impl<T: Access> InsertOnlyMap<T>
316/// where
317///     T::Base: RawAccessMut,
318/// {
319///     fn insert(&mut self, key: &str, value: String) -> bool {
320///         if self.map.contains(key) { return false; }
321///         self.map.put(&key.to_owned(), value);
322///         self.len.set(self.len.get().unwrap_or_default() + 1);
323///         true
324///     }
325/// }
326///
327/// # fn main() -> Result<(), AccessError> {
328/// let db = TemporaryDB::new();
329/// let fork = db.fork();
330/// # {
331/// let mut map = InsertOnlyMap::from_access(&fork, "test".into())?;
332/// map.insert("foo", "FOO".to_owned());
333/// map.insert("bar", "BAR".to_owned());
334/// assert_eq!(map.len.get(), Some(2));
335/// # }
336///
337/// // Components could be used with `Group` / `Lazy` out of the box:
338/// let lazy_map: Lazy<_, InsertOnlyMap<_>> =
339///     Lazy::from_access(&fork, "test".into())?;
340/// assert_eq!(lazy_map.get().map.get("foo").unwrap(), "FOO");
341///
342/// let group_of_maps: Group<_, u16, InsertOnlyMap<_>> =
343///     fork.get_group("test_group");
344/// group_of_maps.get(&1).insert("baz", "BAZ".to_owned());
345/// group_of_maps.get(&2).insert("baz", "BUZZ".to_owned());
346/// # assert_eq!(group_of_maps.get(&1).len.get(), Some(1));
347/// # assert_eq!(
348/// #     fork.get_map::<_, str, String>(("test_group.map", &2_u16)).get("baz").unwrap(),
349/// #     "BUZZ"
350/// # );
351/// # Ok(())
352/// # }
353/// ```
354pub trait FromAccess<T: Access>: Sized {
355    /// Constructs the object at the given address.
356    ///
357    /// # Return value
358    ///
359    /// Returns the constructed object.
360    ///
361    /// # Errors
362    ///
363    /// An error should be returned if the object cannot be constructed.
364    fn from_access(access: T, addr: IndexAddress) -> Result<Self, AccessError>;
365
366    /// Constructs the object from the root of the `access`.
367    ///
368    /// The default implementation uses `Self::from_access()` with an empty address.
369    ///
370    /// # Errors
371    ///
372    /// Returns access errors.
373    fn from_root(access: T) -> Result<Self, AccessError> {
374        Self::from_access(access, IndexAddress::default())
375    }
376}

383    #[test]
384    fn prefixed_works() {
385        let db = TemporaryDB::new();
386        let fork = db.fork();
387        {
388            let prefixed = Prefixed::new("test", &fork);
389            let mut list = prefixed.get_list::<_, i32>("foo");
390            list.extend(vec![1, 2, 3]);
391        }
392        {
393            let list = fork.get_list::<_, i32>("test.foo");
394            assert_eq!(list.len(), 3);
395            assert_eq!(list.iter().collect::<Vec<_>>(), vec![1, 2, 3]);
396        }
397        db.merge_sync(fork.into_patch()).unwrap();
398
399        let snapshot = db.snapshot();
400        let list = snapshot.get_list::<_, i32>("test.foo");
401        assert_eq!(list.len(), 3);
402        assert_eq!(list.iter().collect::<Vec<_>>(), vec![1, 2, 3]);
403
404        let prefixed = Prefixed::new("test", &snapshot);
405        let list = prefixed.get_list::<_, i32>("foo");
406        assert_eq!(list.len(), 3);
407        assert_eq!(list.iter().collect::<Vec<_>>(), vec![1, 2, 3]);
408    }

410    #[test]
411    fn prefixed_views_do_not_collide() {
412        let db = TemporaryDB::new();
413        let fork = db.fork();
414        let foo_space = Prefixed::new("foo", &fork);
415        let bar_space = Prefixed::new("bar", &fork);
416        {
417            let mut list = foo_space.get_list("test");
418            list.push("Test".to_owned());
419            let mut other_list = bar_space.get_list("test");
420            other_list.extend(vec![1_u64, 2, 3]);
421
422            assert_eq!(list.len(), 1);
423            assert_eq!(other_list.len(), 3);
424        }
425        db.merge_sync(fork.into_patch()).unwrap();
426
427        let snapshot = db.snapshot();
428        let foo_space = Prefixed::new("foo", &snapshot);
429        let list = foo_space.get_list::<_, String>("test");
430        assert_eq!(list.get(0), Some("Test".to_owned()));
431        let bar_space = Prefixed::new("bar", &snapshot);
432        let list = bar_space.get_list::<_, u64>("test");
433        assert_eq!(list.get(0), Some(1_u64));
434
435        // It is possible to create indexes of the different types at the same (relative) address
436        // in the different `Prefixed` instances.
437        let fork = db.fork();
438        let foo_space = Prefixed::new("foo", &fork);
439        foo_space.get_list::<_, u32>(("fam", &1_u32));
440        let bar_space = Prefixed::new("bar", &fork);
441        bar_space.get_map::<_, u32, u32>(("fam", &1_u32));
442        db.merge_sync(fork.into_patch()).unwrap();
443
444        let snapshot = db.snapshot();
445        let view = snapshot
446            .as_ref()
447            .get_or_create_view(("foo.fam", &1_u32).into(), IndexType::List)
448            .unwrap();
449        assert!(!view.is_phantom());
450        let view = snapshot
451            .as_ref()
452            .get_or_create_view(("bar.fam", &1_u32).into(), IndexType::Map)
453            .unwrap();
454        assert!(!view.is_phantom());
455    }

458    fn from_root_method() {
459        let db = TemporaryDB::new();
