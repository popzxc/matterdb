6/// Options for the database.
7///
8/// These parameters apply to the underlying database, currently `RocksDB`.
9#[derive(Debug, Clone, Copy, Serialize, Deserialize, PartialEq)]
10#[non_exhaustive]
11pub struct DBOptions {
12    /// Number of open files that can be used by the database.
13    ///
14    /// The underlying database opens multiple files during operation. If your system has a
15    /// limit on the number of files which can be open simultaneously, you can
16    /// adjust this option to match the limit. Note, that limiting the number
17    /// of simultaneously open files might slow down the speed of database operation.
18    ///
19    /// Defaults to `None`, meaning that the number of open files is unlimited.
20    pub max_open_files: Option<i32>,
21    /// An option to indicate whether the system should create a database or not,
22    /// if it's missing.
23    ///
24    /// This option applies to the cases when a node was
25    /// switched off and is on again. If the database cannot be found at the
26    /// indicated path and this option is switched on, a new database will be
27    /// created at that path and blocks will be included therein.
28    ///
29    /// Defaults to `true`.
30    pub create_if_missing: bool,
31    /// An algorithm used for database compression.
32    ///
33    /// Defaults to `CompressionType::None`, meaning there is no compression.
34    pub compression_type: CompressionType,
35    /// Max total size of the WAL journal in bytes.
36    ///
37    /// Defaults to `None`, meaning that the size of WAL journal will be adjusted
38    /// by the rocksdb.
39    pub max_total_wal_size: Option<u64>,
40    /// Max `LRU` in-memory cache size in bytes.
41    ///
42    /// Defaults to `None`, meaning that there will be no cache used.
43    pub max_cache_size: Option<usize>,
44}
45
46impl DBOptions {
47    /// Creates a new `DBOptions` object.
48    pub fn new(
49        max_open_files: Option<i32>,
50        create_if_missing: bool,
51        compression_type: CompressionType,
52        max_total_wal_size: Option<u64>,
53        max_cache_size: Option<usize>,
54    ) -> Self {
55        Self {
56            max_open_files,
57            create_if_missing,
58            compression_type,
59            max_total_wal_size,
60            max_cache_size,
61        }
62    }
63}
64
65/// Algorithms of compression for the database.
66///
67/// Database contents are stored in a set of blocks, each of which holds a
68/// sequence of key-value pairs. Each block may be compressed before
69/// being stored in a file. The following enum describes which
70/// compression algorithm (if any) is used to compress a block.
71#[derive(Debug, Clone, Copy, Serialize, Deserialize, PartialEq)]
72#[serde(rename_all = "lowercase")]
73pub enum CompressionType {
74    Bz2,
75    Lz4,
76    Lz4hc,
77    Snappy,
78    Zlib,
79    Zstd,
80    None,
81}
82
83impl From<CompressionType> for DBCompressionType {
84    fn from(compression_type: CompressionType) -> Self {
85        match compression_type {
86            CompressionType::Bz2 => Self::Bz2,
87            CompressionType::Lz4 => Self::Lz4,
88            CompressionType::Lz4hc => Self::Lz4hc,
89            CompressionType::Snappy => Self::Snappy,
90            CompressionType::Zlib => Self::Zlib,
91            CompressionType::Zstd => Self::Zstd,
92            CompressionType::None => Self::None,
93        }
94    }
95}
96
97impl Default for DBOptions {
98    fn default() -> Self {
99        Self::new(None, true, CompressionType::None, None, None)
100    }
