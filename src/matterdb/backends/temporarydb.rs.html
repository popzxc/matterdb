18type MemoryDB = HashMap<ResolvedAddress, BTreeMap<Vec<u8>, Vec<u8>>>;
19
20/// This in-memory database is only used for testing and experimenting; is not designed to
21/// operate under load in production.
22#[derive(Debug)]
23pub struct TemporaryDB {
24    inner: Arc<ShardedLock<MemoryDB>>,
25}
26
27struct TemporarySnapshot {
28    snapshot: MemoryDB,
29}
30
31struct TemporaryDBIterator<'a> {
32    iter: Peekable<Range<'a, Vec<u8>, Vec<u8>>>,
33    prefix: Option<[u8; ID_SIZE]>,
34    ended: bool,
35}
36
37impl TemporaryDB {
38    /// Creates a new, empty database.
39    #[allow(clippy::missing_panics_doc)] // false positive
40    pub fn new() -> Self {
41        let mut db = HashMap::new();
42
43        db.insert(ResolvedAddress::system("default"), BTreeMap::new());
44        let inner = Arc::new(ShardedLock::new(db));
45        let mut db = Self { inner };
46        check_database(&mut db).unwrap();
47        db
48    }
49
50    /// Clears the contents of the database.
51    #[allow(clippy::missing_panics_doc, clippy::missing_errors_doc)] // FIXME: always returns Ok(())
52    pub fn clear(&self) -> crate::Result<()> {
53        let mut rw_lock = self.inner.write().expect("Couldn't get read-write lock");
54
55        for collection in rw_lock.values_mut() {
56            collection.clear();
57        }
58
59        Ok(())
60    }
61
62    fn temporary_snapshot(&self) -> TemporarySnapshot {
63        TemporarySnapshot {
64            snapshot: self.inner.read().expect("Couldn't get read lock").clone(),
65        }
66    }
67}
68
69impl Database for TemporaryDB {
70    fn snapshot(&self) -> Box<dyn Snapshot> {
71        Box::new(self.temporary_snapshot())
72    }
73
74    fn merge(&self, patch: Patch) -> Result<()> {
75        let mut inner = self.inner.write().expect("Couldn't get write lock");
76        for (resolved, changes) in patch.into_changes() {
77            if !inner.contains_key(&resolved) {
78                inner.insert(resolved.clone(), BTreeMap::new());
79            }
80
81            let collection: &mut BTreeMap<Vec<u8>, Vec<u8>> = inner.get_mut(&resolved).unwrap();
82
83            if changes.is_cleared() {
84                if let Some(id_bytes) = resolved.id_to_bytes() {
85                    let next_bytes = next_id_bytes(id_bytes);
86                    let mut middle_and_tail = collection.split_off(id_bytes.as_ref());
87                    let mut tail = middle_and_tail.split_off(next_bytes.as_ref());
88                    collection.append(&mut tail);
89                } else {
90                    collection.clear();
91                }
92            }
93
94            if let Some(id_bytes) = resolved.id_to_bytes() {
95                // Write changes to the column family with each key prefixed by the ID of the
96                // resolved address.
97
98                // We assume that typical key sizes are less than `1_024 - ID_SIZE = 1_016` bytes,
99                // so that they fit into stack.
100                let mut buffer: SmallVec<[u8; 1_024]> = SmallVec::new();
101                buffer.extend_from_slice(&id_bytes);
102
103                for (key, change) in changes.into_data() {
104                    buffer.truncate(ID_SIZE);
105                    buffer.extend_from_slice(&key);
106
107                    match change {
108                        Change::Put(value) => collection.insert(buffer.to_vec(), value),
109                        Change::Delete => collection.remove(buffer.as_ref()),
110                    };
111                }
112            } else {
113                // Write changes to the column family as-is.
114                for (key, change) in changes.into_data() {
115                    match change {
116                        Change::Put(value) => collection.insert(key, value),
117                        Change::Delete => collection.remove(&key),
118                    };
119                }
120            }
121        }
122        Ok(())
123    }
124
125    fn merge_sync(&self, patch: Patch) -> Result<()> {
126        self.merge(patch)
127    }
128}

130impl DBIterator for TemporaryDBIterator<'_> {
131    fn next(&mut self) -> Option<(&[u8], &[u8])> {
132        if self.ended {
133            return None;
134        }
135
136        let (key, value) = self.iter.next()?;
137
138        if let Some(ref prefix) = self.prefix {
139            if &key[..ID_SIZE] != prefix {
140                self.ended = true;
141                return None;
142            }
143        }
144
145        let key = if self.prefix.is_some() {
146            &key[ID_SIZE..]
147        } else {
148            &key[..]
149        };
150
151        Some((key, value))
152    }
153
154    fn peek(&mut self) -> Option<(&[u8], &[u8])> {
155        if self.ended {
156            return None;
157        }
158
159        let (key, value) = self.iter.peek()?;
160        let key = if let Some(prefix) = self.prefix {
161            if key[..ID_SIZE] != prefix {
162                self.ended = true;
163                return None;
164            }
165            &key[ID_SIZE..]
166        } else {
167            &key[..]
168        };
169
170        Some((key, value))
171    }
172}

174impl Snapshot for TemporarySnapshot {
175    fn get(&self, name: &ResolvedAddress, key: &[u8]) -> Option<Vec<u8>> {
176        let collection = self.snapshot.get(name)?;
177        collection.get(name.keyed(key).as_ref()).cloned()
178    }
179
180    fn iter(&self, name: &ResolvedAddress, from: &[u8]) -> Iter<'_> {
181        let collection = self
182            .snapshot
183            .get(name)
184            .or_else(|| self.snapshot.get(&ResolvedAddress::system("default")))
185            .unwrap();
186        let from = name.keyed(from).into_owned();
187        let iter = collection.range::<Vec<u8>, _>(&from..);
188
189        Box::new(TemporaryDBIterator {
190            iter: iter.peekable(),
191            prefix: name.id_to_bytes(),
192            ended: false,
193        })
194    }
195}
196
197impl Default for TemporaryDB {
198    fn default() -> Self {
199        Self::new()
200    }
201}

210#[test]
211fn clearing_database() {
212    use crate::access::CopyAccessExt;
213
214    let db = TemporaryDB::new();
215    let fork = db.fork();
216
217    fork.get_list("foo").extend(vec![1_u32, 2, 3]);
218    fork.get_entry(("bar", &0_u8)).set("!".to_owned());
219    fork.get_entry(("bar", &1_u8)).set("?".to_owned());
220    db.merge(fork.into_patch()).unwrap();
221    db.clear().unwrap();
222
223    let fork = db.fork();
224
225    assert!(fork.index_type("foo").is_none());
226    assert!(fork.index_type(("bar", &0_u8)).is_none());
227    assert!(fork.index_type(("bar", &1_u8)).is_none());
228    fork.get_list("foo").extend(vec![4_u32, 5, 6]);
229    db.merge(fork.into_patch()).unwrap();
230
231    let snapshot = db.snapshot();
232    let list = snapshot.get_list::<_, u32>("foo");
233
234    assert_eq!(list.len(), 3);
235    assert_eq!(list.iter().collect::<Vec<_>>(), vec![4, 5, 6]);
236}
