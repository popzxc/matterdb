25/// Size of a byte representation of an index ID, which is used to prefix index keys
26/// in a column family.
27pub const ID_SIZE: usize = mem::size_of::<u64>();

29/// Database implementation on top of [`RocksDB`](https://rocksdb.org)
30/// backend.
31///
32/// `RocksDB` is an embedded database for key-value data, which is optimized for fast storage.
33/// This structure is required to potentially adapt the interface to
34/// use different databases.
35#[derive(Clone)]
36pub struct RocksDB {
37    db: Arc<ShardedLock<rocksdb::DB>>,
38    options: DBOptions,
39}

47impl From<&DBOptions> for RocksDBOptions {
48    fn from(opts: &DBOptions) -> Self {
49        let mut defaults = Self::default();
50        defaults.create_if_missing(opts.create_if_missing);
51        defaults.set_compression_type(opts.compression_type.into());
52        defaults.set_max_open_files(opts.max_open_files.unwrap_or(-1));
53        defaults.set_max_total_wal_size(opts.max_total_wal_size.unwrap_or(0));
54        if let Some(capacity) = opts.max_cache_size {
55            defaults.set_row_cache(&RocksDBCache::new_lru_cache(capacity));
56        }
57        defaults
58    }
59}

67/// An iterator over the entries of a `RocksDB`.
68struct RocksDBIterator<'a> {
69    iter: Peekable<DBIterator<'a>>,
70    key: Option<Box<[u8]>>,
71    value: Option<Box<[u8]>>,
72    prefix: Option<[u8; ID_SIZE]>,
73    ended: bool,
74}

76impl RocksDB {
77    /// Opens a database stored at the specified path with the specified options.
78    ///
79    /// If the database does not exist at the indicated path and the option
80    /// `create_if_missing` is switched on in `DBOptions`, a new database will
81    /// be created at the indicated path.
82    ///
83    /// # Errors
84    ///
85    /// Propagates I/O errors. Returns an error on incompatible MatterDB version.
86    pub fn open<P: AsRef<Path>>(path: P, options: &DBOptions) -> crate::Result<Self> {
87        let inner = {
88            if let Ok(names) = rocksdb::DB::list_cf(&RocksDBOptions::default(), &path) {
89                let cf_names = names.iter().map(String::as_str).collect::<Vec<_>>();
90                rocksdb::DB::open_cf(&options.into(), path, cf_names)?
91            } else {
92                rocksdb::DB::open(&options.into(), path)?
93            }
94        };
95        let mut db = Self {
96            db: Arc::new(ShardedLock::new(inner)),
97            options: *options,
98        };
99        check_database(&mut db)?;
100        Ok(db)
101    }
102
103    /// Creates checkpoint of this database in the given directory. See [`RocksDB` docs] for
104    /// details.
105    ///
106    /// Successfully created checkpoint can be opened using `RocksDB::open`.
107    ///
108    /// [`RocksDB` docs]: https://github.com/facebook/rocksdb/wiki/Checkpoints
109    ///
110    /// # Errors
111    ///
112    /// Propagates I/O errors.
113    pub fn create_checkpoint<T: AsRef<Path>>(&self, path: T) -> crate::Result<()> {
114        let guard = self.get_db_lock_guard();
115        let checkpoint = Checkpoint::new(&*guard)?;
116        checkpoint.create_checkpoint(path)?;
117        Ok(())
118    }

120    /// Retrieves read lock guard containing underlying `rocksdb::DB`.
121    #[allow(clippy::missing_panics_doc)] // FIXME: make private?
122    pub fn get_db_lock_guard(&self) -> ShardedLockReadGuard<'_, rocksdb::DB> {
123        self.db.read().expect("Failed to get read lock to DB")
124    }

126    fn cf_exists(&self, cf_name: &str) -> bool {
127        self.get_db_lock_guard().cf_handle(cf_name).is_some()
128    }
129
130    fn create_cf(&self, cf_name: &str) -> crate::Result<()> {
131        self.db
132            .write()
133            .expect("Failed to get write lock to DB")
134            .create_cf(cf_name, &self.options.into())
135            .map_err(Into::into)
136    }

138    /// Clears the column family completely, removing all keys from it.
139    pub(super) fn clear_column_family(&self, batch: &mut WriteBatch, cf: &ColumnFamily) {
140        /// Some lexicographically large key.
141        const LARGER_KEY: &[u8] = &[u8::MAX; 1_024];
142
143        let db_reader = self.get_db_lock_guard();
144        let mut iter = db_reader.raw_iterator_cf(cf);
145        iter.seek_to_last();
146        if iter.valid() {
147            if let Some(key) = iter.key() {
148                // For some reason, removing a range to a very large key is
149                // significantly faster than removing the exact range.
150                // This is specific to the debug mode, but since `TemporaryDB`
151                // is mostly used for testing, this optimization leads to practical
152                // performance improvement.
153                if key.len() < LARGER_KEY.len() {
154                    batch.delete_range_cf(cf, &[][..], LARGER_KEY);
155                } else {
156                    batch.delete_range_cf(cf, &[][..], key);
157                    batch.delete_cf(cf, key);
158                }
159            }
160        }
161    }

163    fn do_merge(&self, patch: Patch, w_opts: &RocksDBWriteOptions) -> crate::Result<()> {
164        let mut batch = WriteBatch::default();
165        for (resolved, changes) in patch.into_changes() {
166            if !self.cf_exists(&resolved.name) {
167                self.create_cf(&resolved.name)?;
168            }
169
170            let db_reader = self.get_db_lock_guard();
171            let cf = db_reader.cf_handle(&resolved.name).unwrap();
172
173            if changes.is_cleared() {
174                self.clear_prefix(&mut batch, cf, &resolved);
175            }
176
177            if let Some(id_bytes) = resolved.id_to_bytes() {
178                // Write changes to the column family with each key prefixed by the ID of the
179                // resolved address.
180
181                // We assume that typical key sizes are less than `1_024 - ID_SIZE = 1_016` bytes,
182                // so that they fit into stack.
183                let mut buffer: SmallVec<[u8; 1_024]> = SmallVec::new();
184                buffer.extend_from_slice(&id_bytes);
185
186                for (key, change) in changes.into_data() {
187                    buffer.truncate(ID_SIZE);
188                    buffer.extend_from_slice(&key);
189                    match change {
190                        Change::Put(ref value) => batch.put_cf(cf, &buffer, value),
191                        Change::Delete => batch.delete_cf(cf, &buffer),
192                    }
193                }
194            } else {
195                // Write changes to the column family as-is.
196                for (key, change) in changes.into_data() {
197                    match change {
198                        Change::Put(ref value) => batch.put_cf(cf, &key, value),
199                        Change::Delete => batch.delete_cf(cf, &key),
200                    }
201                }
202            }
203        }
204
205        self.get_db_lock_guard()
206            .write_opt(batch, w_opts)
207            .map_err(Into::into)
208    }
209
210    /// Removes all keys with the specified prefix from a column family.
211    fn clear_prefix(&self, batch: &mut WriteBatch, cf: &ColumnFamily, resolved: &ResolvedAddress) {
212        if let Some(id_bytes) = resolved.id_to_bytes() {
213            let next_bytes = next_id_bytes(id_bytes);
214            batch.delete_range_cf(cf, id_bytes, next_bytes);
215        } else {
216            self.clear_column_family(batch, cf);
217        }
218    }

220    #[allow(unsafe_code)]
221    pub(super) fn rocksdb_snapshot(&self) -> RocksDBSnapshot {
222        RocksDBSnapshot {
223            // SAFETY:
224            // The snapshot carries an `Arc` to the database to make sure that database
225            // is not dropped before the snapshot. Additionally, the pointer to `rocksdb::DB`
226            // is stable within `Arc<ShardedLock<rocksdb::DB>>` and its part used in dropping
227            // the snapshot (`*mut ffi::rocksdb_t`) is never changed, i.e., not affected
228            // by potential incoherence if the `ShardedLock` is being concurrently written to.
229            // FIXME: Investigate changing `rocksdb::Snapshot` / `DB` to remove `unsafe` (ECR-4273).
230            snapshot: unsafe {
231                mem::transmute::<rocksdb::Snapshot<'_>, rocksdb::Snapshot<'static>>(
232                    self.get_db_lock_guard().snapshot(),
233                )
234            },
235            db: Arc::clone(&self.db),
236        }
237    }

240impl RocksDBSnapshot {
241    fn get_lock_guard(&self) -> ShardedLockReadGuard<'_, rocksdb::DB> {
242        self.db.read().expect("Failed to get read lock to DB")
243    }
244
245    fn rocksdb_iter(&self, name: &ResolvedAddress, from: &[u8]) -> RocksDBIterator<'_> {
246        use rocksdb::{Direction, IteratorMode};
247
248        let from = name.keyed(from);
249        let iter = match self.get_lock_guard().cf_handle(&name.name) {
250            Some(cf) => self
251                .snapshot
252                .iterator_cf(cf, IteratorMode::From(from.as_ref(), Direction::Forward)),
253            None => self.snapshot.iterator(IteratorMode::Start),
254        };
255        RocksDBIterator {
256            iter: iter.peekable(),
257            prefix: name.id_to_bytes(),
258            key: None,
259            value: None,
260            ended: false,
261        }
262    }
263}

282impl Snapshot for RocksDBSnapshot {
283    fn get(&self, resolved_addr: &ResolvedAddress, key: &[u8]) -> Option<Vec<u8>> {
284        let lock = self.get_lock_guard();
285        let cf = lock.cf_handle(&resolved_addr.name)?;
286        self.snapshot
287            .get_cf(cf, resolved_addr.keyed(key))
288            .unwrap_or_else(|e| panic!("{}", e))
289    }
290
291    fn iter(&self, name: &ResolvedAddress, from: &[u8]) -> Iter<'_> {
292        Box::new(self.rocksdb_iter(name, from))
293    }

296impl Iterator for RocksDBIterator<'_> {
297    fn next(&mut self) -> Option<(&[u8], &[u8])> {
298        if self.ended {
299            return None;
300        }
301
302        let (key, value) = self.iter.next()?.expect("failed iterating over RocksDB");
303        if let Some(ref prefix) = self.prefix {
304            if &key[..ID_SIZE] != prefix {
305                self.ended = true;
306                return None;
307            }
308        }
309
310        self.key = Some(key);
311        let key = if self.prefix.is_some() {
312            &self.key.as_ref()?[ID_SIZE..]
313        } else {
314            &self.key.as_ref()?[..]
315        };
316        self.value = Some(value);
317        Some((key, self.value.as_ref()?))
318    }
319
320    fn peek(&mut self) -> Option<(&[u8], &[u8])> {
321        if self.ended {
322            return None;
323        }
324
325        let (key, value) = self
326            .iter
327            .peek()?
328            .as_ref()
329            .expect("failed iterating over RocksDB");
330        let key = if let Some(prefix) = self.prefix {
331            if key[..ID_SIZE] != prefix {
332                self.ended = true;
333                return None;
334            }
335            &key[ID_SIZE..]
336        } else {
337            &key[..]
338        };
339        Some((key, &value[..]))
340    }
341}

361/// Generates the sequence of bytes lexicographically following the provided one. Assumes that
362/// the provided sequence is less than `[u8::MAX; ID_SIZE]`.
363pub fn next_id_bytes(id_bytes: [u8; ID_SIZE]) -> [u8; ID_SIZE] {
364    let mut next_id_bytes = id_bytes;
365    for byte in next_id_bytes.iter_mut().rev() {
366        if *byte == u8::MAX {
367            *byte = 0;
368        } else {
369            *byte += 1;
370            break;
371        }
372    }
373    next_id_bytes
374}

376#[test]
377fn test_next_id_bytes() {
378    assert_eq!(
379        next_id_bytes([1, 0, 0, 0, 0, 0, 0, 0]),
380        [1, 0, 0, 0, 0, 0, 0, 1]
381    );
382    assert_eq!(
383        next_id_bytes([1, 2, 3, 4, 5, 6, 7, 8]),
384        [1, 2, 3, 4, 5, 6, 7, 9]
385    );
386    assert_eq!(
387        next_id_bytes([1, 0, 0, 0, 0, 0, 0, 254]),
388        [1, 0, 0, 0, 0, 0, 0, 255]
389    );
390    assert_eq!(
391        next_id_bytes([1, 0, 0, 0, 0, 0, 41, 255]),
392        [1, 0, 0, 0, 0, 0, 42, 0]
393    );
394    assert_eq!(
395        next_id_bytes([1, 2, 3, 4, 5, 255, 255, 255]),
396        [1, 2, 3, 4, 6, 0, 0, 0]
397    );
