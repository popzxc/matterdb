1//! Migration utilities.
2//!
3//! # Stability
4//!
5//! The entirety of this module is considered unstable. While the supported functionality
6//! is unlikely to break, the implementation details may change in the following releases.
7//!
8//! # Migration workflow
9//!
10//! **Migration** refers to the ability to update data in indexes, remove indexes,
11//! change index type, create new indexes, and package these changes in a way that they
12//! can be atomically committed or rolled back. Accumulating changes in the migration,
13//! on the other hand, can be performed iteratively, including after a process shutdown.
14//!
15//! Each migration is confined to a *namespace*, defined in a similar way as [`Prefixed`]
16//! accesses. For example, namespace `test` concerns indexes with an address starting with
17//! `test.`, such as `test.foo` or `(test.bar, 1_u32)`, but not `test` or `test_.foo`.
18//! The namespace can be accessed via [`Migration`].
19//!
20//! Migration is non-destructive, i.e., does not remove the old versions of migrated indexes.
21//! Instead, new indexes are created in a separate namespace. For example, index `foo`
22//! in the migration namespace `test` and the original `test.foo` index can peacefully coexist
23//! and have separate data and even different types. The movement of data is performed only
24//! when the migration is finalized.
25//!
26//! Retaining an index in the migration is a no op. *Removing* an index is explicit; it needs
27//! to be performed via [`create_tombstone`] method. Although tombstones do not contain data,
28//! they behave like indexes in other regards. For example, it is impossible to create a tombstone
29//! and then create an ordinary index at the same address, or vice versa.
30//!
31//! A migration can also store temporary data in a [`Scratchpad`]. This data will be removed
32//! when the migration is finalized.
33//!
34//! Indexes created within a migration are not [aggregated] in the default state hash. Instead,
35//! they are placed in a separate namespace, the aggregator and state hash for which can be
36//! obtained via respective [`Migration`] methods.
37//!
38//! It is possible to periodically persist migrated data to the database
39//! (indeed, this is a best practice to avoid out-of-memory errors). It is even possible
40//! to restart the process handling the migration, provided it can recover from such a restart
41//! on the application level. To assist with fault tolerance, use [persistent iterators].
42//!
43//! # Finalizing Migration
44//!
45//! To finalize a migration, one needs to call [`flush_migration`]. This will replace
46//! old index data with new, remove indexes marked with tombstones, and return migrated indexes
47//! to the default state aggregator. To roll back a migration,
48//! use [`rollback_migration`]. This will remove the new index data and corresponding metadata.
49//! Both `flush_migration` and `rollback_migration` will remove the `Scratchpad` associated
50//! with the migration.
51//!
52//! [`Migration`]: struct.Migration.html
53//! [`Prefixed`]: ../access/struct.Prefixed.html
54//! [`create_tombstone`]: struct.Migration.html#method.create_tombstone
55//! [`Scratchpad`]: struct.Scratchpad.html
56//! [aggregated]: ../index.html#state-aggregation
57//! [persistent iterators]: struct.PersistentIter.html
58//! [`flush_migration`]: fn.flush_migration.html
59//! [`rollback_migration`]: fn.rollback_migration.html
60//!
61//! # Examples
62//!
63//! None yet.

91/// Access to migrated indexes.
92///
93/// `Migration` is conceptually similar to a [`Prefixed`] access. For example, an index with
94/// address `"list"` in a migration `Migration::new("foo", _)` will map to the address `"foo.list"`
95/// after the migration is flushed. The major difference with `Prefixed` is that the indexes
96/// in a migration cannot be accessed in any other way. That is, it is impossible to access
97/// an index in a migration without constructing a `Migration` object first.
98///
99/// [`Prefixed`]: ../access/struct.Prefixed.html
100#[derive(Debug, Clone)]
101pub struct Migration<T> {
102    access: T,
103    namespace: String,
104}
105
106// **NB.** Must not be made public! This would allow the caller to violate access restrictions
107// imposed by `Migration`.
108impl<T> Migration<T> {
109    pub(crate) fn access(&self) -> &T {
110        &self.access
111    }
112
113    pub(crate) fn into_parts(self) -> (String, T) {
114        (self.namespace, self.access)
115    }
116}
117
118impl<T: RawAccess> Migration<T> {
119    /// Creates a migration in the specified namespace.
120    pub fn new(namespace: impl Into<String>, access: T) -> Self {
121        Self {
122            namespace: namespace.into(),
123            access,
124        }
125    }
126}

128impl<T: RawAccessMut> Migration<T> {
129    /// Marks an index with the specified address as removed during migration.
130    ///
131    /// # Panics
132    ///
133    /// Panics if an index already exists at the specified address.
134    pub fn create_tombstone<I>(&self, addr: I)
135    where
136        I: Into<IndexAddress>,
137    {
138        self.clone()
139            .get_or_create_view(addr.into(), IndexType::Tombstone)
140            .unwrap_or_else(|e| panic!("MerkleDB error: {e}"));
141    }
142}

148        let mut prefixed_addr = addr.prepend_name(&self.namespace);

169        prefixed_addr.set_in_migration();
170        self.access.group_keys(prefixed_addr)
171    }
172}

174/// Access to temporary data that can be used during migration. The scratchpad is cleared
175/// at the end of the migration, regardless of whether the migration is successful.
176///
177/// Like `Migration`s, `Scratchpad`s are separated via namespaces. Scratchpads are optimized
178/// for small amounts of data per index. Indexes in a `Scratchpad` are not aggregated into
179/// the overall database state or the migration state.
180#[derive(Debug, Clone)]
181pub struct Scratchpad<T> {
182    access: T,
183    namespace: String,
184}
185
186// **NB.** Must not be made public! This would allow the caller to violate access restrictions
187// imposed by `Scratchpad`.
188impl<T> Scratchpad<T> {
189    pub(crate) fn access(&self) -> &T {
190        &self.access
191    }
192
193    pub(crate) fn into_parts(self) -> (String, T) {
194        (self.namespace, self.access)
195    }
196}
197
198impl<T: RawAccess> Scratchpad<T> {
199    /// Creates a scratchpad in the specified namespace.
200    pub fn new(namespace: impl Into<String>, access: T) -> Self {
201        Self {
202            namespace: namespace.into(),
203            access,
204        }
205    }
206
207    fn get_scratchpad_addr(&self, addr: IndexAddress) -> IndexAddress {
208        let prefixed_addr = addr.prepend_name(&self.namespace);
209        IndexAddress::from_root(SCRATCHPAD_NAME).append_key(&prefixed_addr.fully_qualified_name())
210    }
211
212    fn get_scratchpad_prefix(&self, addr: IndexAddress) -> IndexAddress {
213        let prefixed_addr = addr.prepend_name(&self.namespace);
214        IndexAddress::from_root(SCRATCHPAD_NAME).append_key(&prefixed_addr.qualified_prefix())
215    }
216}
217
218impl<T: RawAccessMut> Scratchpad<T> {
219    /// Removes all indexes and their data from the scratchpad.
220    ///
221    /// # Panics
222    ///
223    /// This operation will panic if any of the removed indexes are borrowed.
224    fn clear(&self) {
225        let addr = self.get_scratchpad_addr(IndexAddress::default());
226        let addr = addr.append_key(&b'.');
227        let removed = IndexesPool::new(self.access.clone()).remove_indexes(&addr);
228        for resolved_addr in removed {
229            View::new(self.access.clone(), resolved_addr).clear();
230        }
231    }
232}
233
234impl<T: RawAccess> Access for Scratchpad<T> {
235    type Base = T;
236
237    fn get_index_metadata(self, addr: IndexAddress) -> Result<Option<IndexMetadata>, AccessError> {
238        let addr = self.get_scratchpad_addr(addr);
239        Ok(ViewWithMetadata::get_metadata_unchecked(self.access, &addr))
240    }
241
242    fn get_or_create_view(
243        self,
244        addr: IndexAddress,
245        index_type: IndexType,
246    ) -> Result<ViewWithMetadata<Self::Base>, AccessError> {
247        // Since we transform the address into `id_in_group`, we need to ensure that addresses
248        // cannot alias each other. We do this by running the sanity check on the original address.
249        if let Err(kind) = check_index_valid_full_name(addr.name()) {
250            return Err(AccessError { addr, kind });
251        }
252        let addr = self.get_scratchpad_addr(addr);
253        ViewWithMetadata::get_or_create_unchecked(self.access, &addr, index_type)
254    }
255
256    fn group_keys<K>(self, base_addr: IndexAddress) -> GroupKeys<Self::Base, K>
257    where
258        K: BinaryKey + ?Sized,
259        Self::Base: AsReadonly<Readonly = Self::Base>,
260    {
261        let base_addr = self.get_scratchpad_prefix(base_addr);
262        self.access.group_keys(base_addr)
263    }
264}

266/// Migration helper.
267///
268/// # Examples
269///
270/// See the [module docs](index.html) for a basic example of usage.
271///
272/// ## Aborting migration
273///
274/// `MigrationHelper` offers [`AbortHandle`] to abort migration logic. Once aborted, `MigrationHelper`
275/// does not allow to merge changes to the database; the relevant methods will return
276/// [`MigrationError::Aborted`]. This is important, e.g., to prevent unnecessary writes
277/// to the database.
278///
279/// [`AbortHandle`]: struct.AbortHandle.html
280/// [`MigrationError::Aborted`]: enum.MigrationError.html#variant.Aborted
281///
282/// ```
283/// # use assert_matches::assert_matches;
284/// # use matterdb::{access::CopyAccessExt, TemporaryDB};
285/// # use matterdb::migration::{MigrationHelper, MigrationError};
286/// # use std::{sync::mpsc, thread, time::Duration};
287/// let db = TemporaryDB::new();
288/// // Since `MigrationHelper` cannot be sent between threads, we instantiate it
289/// // in a newly spawned thread and move the helper handle to the main thread.
290/// let (tx, rx) = mpsc::channel();
291/// let helper_thread = thread::spawn(move || {
292///     let (mut helper, handle) = MigrationHelper::with_handle(db, "test");
293///     tx.send(handle).unwrap();
294///     // Emulate some work...
295///     thread::sleep(Duration::from_millis(50));
296///     // Attempt to merge changes to DB.
297///     helper.merge()
298/// });
299///
300/// let handle = rx.recv().unwrap();
301/// // Migration is automatically aborted when the handle is dropped.
302/// drop(handle);
303/// let res: Result<(), MigrationError> = helper_thread.join().unwrap();
304/// assert_matches!(res, Err(MigrationError::Aborted));
305/// ```
306///
307// TODO: The following section was left because I'm not sure what to do with it right now.
308// ## Using persistent iterators
309//
310// `MigrationHelper` offers the [`iter_loop`](#method.iter_loop) method, which allows to further
311// simplify working with [persistent iterators].
312//
313// Say we want to migrate `MapIndex` data to a `ProofMapIndex` while merging changes to the DB
314// from time to time. To do this, we use the following script:
315//
316// ```
317// # use matterdb::{access::AccessExt, TemporaryDB};
318// # use matterdb::migration::{MigrationHelper, MigrationError};
319// # fn main() -> Result<(), MigrationError> {
320// /// Number of accounts processed per DB merge.
321// const CHUNK_SIZE: usize = 100;
322//
323// let db = TemporaryDB::new();
324// let mut helper = MigrationHelper::new(db, "test");
325// helper.iter_loop(|helper, iters| {
326//     // The data before migration is stored in this map
327//     let old_map = helper.old_data().get_map::<_, str, u64>("wallets");
328//     // ...and the new data is in this merkelized map.
329//     let mut new_map = helper.new_data().get_map::<_, str, u64>("wallets");
330//
331//     // Create an iterator over the old data.
332//     let iter = iters.create("wallets", &old_map);
333//     // Take a fixed amount of records from the iterator and migrate them.
334//     // Since `iter` is persistent, it will not return the same record twice,
335//     // even if this script is restarted.
336//     for (name, balance) in iter.take(CHUNK_SIZE) {
337//         new_map.put(&name, balance);
338//     }
339// })?;
340// // Here, the iterator has run out of items. The script can now perform
341// // other actions if necessary.
342// # Ok(())
343// # }
344// ```
345//
346// [persistent iterators]: struct.PersistentIter.html
347pub struct MigrationHelper {
348    db: Arc<dyn Database>,
349    abort_handle: Box<dyn AbortMigration>,
350    // Only equals `None` during merges.
351    fork: Option<Fork>,
352    namespace: String,
353}
354
355impl fmt::Debug for MigrationHelper {
356    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
357        formatter
358            .debug_tuple("MigrationHelper")
359            .field(&self.namespace)
360            .finish()
361    }
362}

364impl MigrationHelper {
365    /// Creates a new helper.
366    pub fn new(db: impl Into<Arc<dyn Database>>, namespace: &str) -> Self {
367        assert_valid_name_component(namespace);
368
369        let db = db.into();
370        Self {
371            fork: Some(db.fork()),
372            db,
373            abort_handle: Box::new(()),
374            namespace: namespace.to_owned(),
375        }
376    }
377
378    /// Creates a new helper together with the abort handle. Unlike the `MigrationHelper`,
379    /// the handle may be sent between threads. The handle allows to determine whether the migration
380    /// helper was completed, and allows to abort the migration by preventing further writes
381    /// to the database.
382    pub fn with_handle(db: impl Into<Arc<dyn Database>>, namespace: &str) -> (Self, AbortHandle) {
383        let mut this = Self::new(db, namespace);
384        let abort_handle = AbortHandle {
385            inner: Arc::new(AtomicBool::default()),
386        };
387        this.set_abort_handle(abort_handle.clone_inner());
388        (this, abort_handle)
389    }
390
391    /// Sets the abort handle for the helper.
392    ///
393    /// # Stability
394    ///
395    /// This method is considered experimental. Its signature may be changed or it may be removed
396    /// in the future.
397    #[doc(hidden)]
398    pub fn set_abort_handle(&mut self, abort_handle: impl AbortMigration + 'static) {
399        self.abort_handle = Box::new(abort_handle);
400    }
401
402    fn fork_ref(&self) -> &Fork {
403        // `unwrap` is safe due to the way we define `fork`
404        self.fork.as_ref().unwrap()
405    }
406
407    /// Checks if the migration has been aborted.
408    fn is_aborted(&self) -> bool {
409        self.abort_handle.is_aborted()
410    }
411
412    /// Returns full access to the new version of migrated data.
413    pub fn new_data(&self) -> Migration<&Fork> {
414        Migration::new(&self.namespace, self.fork_ref())
415    }
416
417    /// Returns the scratchpad for temporary data to use during migration.
418    pub fn scratchpad(&self) -> Scratchpad<&Fork> {
419        Scratchpad::new(&self.namespace, self.fork_ref())
420    }
421
422    /// Returns readonly access to the old version of migrated data.
423    pub fn old_data(&self) -> Prefixed<ReadonlyFork<'_>> {
424        Prefixed::new(&self.namespace, self.fork_ref().readonly())
425    }
426
427    /// Merges the changes to the migrated data and the scratchpad to the database.
428    ///
429    /// `merge` does not flush the migration; the migrated data remains in a separate namespace.
430    /// Use [`flush_migration`] to flush the migrated data.
431    ///
432    /// [`flush_migration`]: fn.flush_migration.html
433    ///
434    /// # Errors
435    ///
436    /// Returns an error if the migration is aborted or merging the changes into the DB fails.
437    #[allow(clippy::missing_panics_doc)] // false positive
438    pub fn merge(&mut self) -> Result<(), MigrationError> {
439        let fork = self.fork.take().unwrap();
440        let patch = fork.into_patch();
441        if self.is_aborted() {
442            Err(MigrationError::Aborted)
443        } else {
444            self.db.merge(patch).map_err(MigrationError::Merge)?;
445            self.fork = Some(self.db.fork());
446            Ok(())
447        }
448    }
449
450    /// Executes the provided closure in a loop until all persistent iterators instantiated
451    /// within the closure have ended. After each iteration, the changes in migrated data are
452    /// merged to the database.
453    ///
454    /// If no iterators are instantiated within the closure, a single iteration will be performed.
455    ///
456    /// # Errors
457    ///
458    /// An error is returned if the DB merge fails on any iteration.
459    pub fn iter_loop(
460        &mut self,
461        mut step: impl FnMut(&Self, &mut PersistentIters<Scratchpad<&Fork>>),
462    ) -> Result<(), MigrationError> {
463        let mut should_break = false;
464        while !should_break {
465            let mut iterators = PersistentIters::new(self.scratchpad());
466            step(self, &mut iterators);
467            should_break = iterators.all_ended();
468            self.merge()?;
469        }
470        Ok(())
471    }
472
473    /// Merges the changes to the migrated data and the migration scratchpad to the database.
474    /// Returns hash representing migrated data state, or an error if the merge has failed.
475    ///
476    /// `finish` does not flush the migration; the migrated data remains in a separate namespace.
477    /// Use [`flush_migration`] to flush the migrated data.
478    ///
479    /// [`flush_migration`]: fn.flush_migration.html
480    ///
481    /// # Errors
482    ///
483    /// Returns an error if the migration is aborted or merging the changes into the DB fails.
484    #[allow(clippy::missing_panics_doc)] // false positive
485    pub fn finish(mut self) -> Result<(), MigrationError> {
486        let patch = self.fork.take().unwrap().into_patch();
487        if self.is_aborted() {
488            Err(MigrationError::Aborted)
489        } else {
490            self.db.merge(patch).map_err(MigrationError::Merge)?;
491            Ok(())
492        }
493    }
494}

496/// Errors emitted by `MigrationHelper` methods.
497#[derive(Debug, Error)]
498pub enum MigrationError {
499    /// Failed to merge migration changes to database.
500    #[error("Failed to merge migration changes to the database: {0}")]
501    Merge(#[source] crate::Error),
502
503    /// Migration has been aborted.
504    #[error("Migration was aborted")]
505    Aborted,
506}

508/// Denotes a communication channel between `MigrationHelper` and the outside world allowing
509/// the helper to understand if the migration is aborted.
510#[doc(hidden)]
511pub trait AbortMigration: Send {
512    /// Has the migration been aborted? `MigrationHelper` calls this method every time before it
513    /// merges changes to the database. If the method returns `true`, the merge is cancelled
514    /// and `MigrationHelper` returns `MigrationError::Aborted`.
515    fn is_aborted(&self) -> bool;
516}
517
518impl AbortMigration for () {
519    fn is_aborted(&self) -> bool {
520        false
521    }
522}
523
524impl AbortMigration for AbortHandle {
525    fn is_aborted(&self) -> bool {
526        self.inner.load(Ordering::SeqCst)
527    }
528}

530/// Handle allowing to signal to `MigrationHelper` that the migration has been aborted.
531/// Signalling is performed on handle drop, unless it is performed with [`forget`](#method.forget)
532/// method.
533#[derive(Debug)]
534pub struct AbortHandle {
535    inner: Arc<AtomicBool>,
536}
537
538impl Drop for AbortHandle {
539    fn drop(&mut self) {
540        self.inner.store(true, Ordering::SeqCst);
541    }
542}
543
544impl AbortHandle {
545    fn clone_inner(&self) -> Self {
546        Self {
547            inner: Arc::clone(&self.inner),
548        }
549    }
550
551    /// Returns `true` if the `MigrationHelper` associated with this handle has been dropped.
552    pub fn is_finished(&self) -> bool {
553        Arc::strong_count(&self.inner) <= 1
554    }
555
556    /// Drops the handle without aborting the migration.
557    pub fn forget(mut self) {
558        self.inner = Arc::new(AtomicBool::default());
559    }
560}

562/// Flushes the migration to the fork. Once the `fork` is merged, the migration is complete.
563///
564/// The following operations will be performed:
565///
566/// - Migrated indexes will replace their old versions
567/// - Migrated indexes will be aggregated in the default namespace
568/// - Indexes marked with tombstones will be removed
569/// - Scratchpad associated with the migration will be cleared
570///
571/// # Safety
572///
573/// Flushing a migration must be performed on a `fork` which contains the final migration
574/// changes. Not doing so **may break the state aggregation in the database.** A scenario when
575/// this requirement would be violated is as follows:
576///
577/// 1. Start a database migration in a separate thread, constructing a `MigrationHelper` around
578///    `Arc<dyn Database>`.
579/// 2. Create a fork.
580/// 3. Ensure that the migration is complete via some synchronization primitive.
581/// 4. Call `flush_migration` on the fork from step 2.
582///
583/// In this scenario, a fork may not have the latest migration data because it was created before
584/// the migration is complete. The correct workflow would be to swap steps 2 and 3, i.e.,
585/// first ensure that the migration is complete and *then* create a fork in which it will be flushed.
586pub fn flush_migration(fork: &mut Fork, namespace: &str) {
587    fork.flush_migration(namespace);
588    Scratchpad::new(namespace, &*fork).clear();
589}
590
591/// Rolls back the migration.
592///
593/// The following operations will be performed:
594///
595/// - Migrated indexes will be erased (both data and metadata)
596/// - Scratchpad associated with the migration will be cleared
597pub fn rollback_migration(fork: &mut Fork, namespace: &str) {
598    fork.rollback_migration(namespace);
599    Scratchpad::new(namespace, &*fork).clear();
600}

678        fn check_indexes<T: RawAccess + Copy>(view: T) {
679            let list = view.get_list::<_, u64>("name.list");
680            assert_eq!(list.len(), 4);
681            assert_eq!(list.get(2), Some(6));
682            assert_eq!(list.iter_from(1).collect::<Vec<_>>(), vec![5, 6, 7]);
683
684            let map = view.get_map::<_, u64, i32>("name.map");
685            assert_eq!(map.get(&1), None);
686            assert_eq!(map.get(&2), Some(21));
687            assert_eq!(map.get(&3), Some(7));
688            assert_eq!(map.keys().collect::<Vec<_>>(), vec![2, 3]);
689
690            // This entry should be removed.
691            let entry = view.get_entry::<_, String>(("name.family", &1_u8));
692            assert!(!entry.exists());
693            // ...but this one should be retained.
694            let entry = view.get_entry::<_, String>(("name.family", &2_u8));
695            assert_eq!(entry.get().unwrap(), "!!");
696
697            let entry = view.get_entry::<_, String>(("name.untouched", &2_u32));
698            assert_eq!(entry.get().unwrap(), "??");
699
700            assert_eq!(view.get_entry("unrelated").get(), Some(1_u64));
701            assert_eq!(view.get_entry("name1.unrelated").get(), Some(2_u64));
702            let set = view.get_key_set::<_, String>("name.removed");
703            assert_eq!(set.iter().count(), 0);
704        }

707

709        fork.get_list("name.list").extend(vec![1_u32, 2, 3]);
710        fork.get_map("name.map").put(&1_u64, "!".to_owned());

752    fn test_migration_rollback(with_merge: bool) {
753        let db = TemporaryDB::new();
754        let mut fork = db.fork();
755
756        fork.get_entry("test.foo").set(1_u8);
757        fork.get_list(("test.list", &1)).extend(vec![1_i32, 2, 3]);
758        let migration = Migration::new("test", &fork);
759        migration.get_entry("foo").set(2_u8);
760        migration.create_tombstone(("list", &1));
761        migration.get_key_set("new").insert(&"test".to_owned());
762
763        if with_merge {
764            db.merge(fork.into_patch()).unwrap();
765            fork = db.fork();
766        }
767        fork.rollback_migration("test");
768        assert_eq!(fork.get_entry::<_, u8>("test.foo").get(), Some(1));
769        let patch = fork.into_patch();
770        assert_eq!(patch.get_entry::<_, u8>("test.foo").get(), Some(1));
771        assert_eq!(
772            patch
773                .get_list::<_, i32>(("test.list", &1))
774                .iter()
775                .collect::<Vec<_>>(),
776            vec![1_i32, 2, 3]
777        );
778
779        let migration = Migration::new("test", &patch);
780        assert!(!migration.get_entry::<_, u8>("foo").exists());
781        // Since migrated indexes don't exist, it should be OK to assign new types to them.
782        assert!(!migration.get_entry::<_, ()>(("list", &1)).exists());
783        assert!(!migration.get_entry::<_, ()>("new").exists());
784    }

964    struct MigrationRig {
965        thread_handle: thread::JoinHandle<Result<(), MigrationError>>,
966        abort_handle: AbortHandle,
967    }
968
969    impl MigrationRig {
970        fn new(db: &Arc<TemporaryDB>) -> Self {
971            let db = Arc::clone(db) as Arc<dyn Database>;
972            let (tx, rx) = mpsc::channel();
973            let thread_handle = thread::spawn(move || {
974                let (helper, handle) = MigrationHelper::with_handle(db, "test");
975                tx.send(handle).unwrap();
976                thread::sleep(Duration::from_millis(50));
977                helper.new_data().get_entry("entry").set(1_u32);
978                helper.finish()
979            });
980
981            Self {
982                thread_handle,
983                abort_handle: rx.recv().unwrap(),
984            }
985        }
986    }
